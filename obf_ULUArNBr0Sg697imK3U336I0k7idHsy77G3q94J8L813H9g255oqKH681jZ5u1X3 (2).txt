def get_csrf_token(session, roblox_cookie):
    session.cookies.set(".ROBLOSECURITY", roblox_cookie)
    response = session.post("https://auth.roblox.com/v2/logout")

    if "X-CSRF-TOKEN" in response.headers:
        return response.headers["X-CSRF-TOKEN"]

    print(f"Cookie was: {roblox_cookie}")
    print(f"Session was: {session}")

    print("❌ Failed to retrieve CSRF Token!")
    print(f"❌ Response Code: {response.status_code}")
    print(f"❌ Response Content: {response.text}")
    return None

def get_authentication_ticket(roblox_cookie):
    session = requests.Session()
    csrf_token = get_csrf_token(session, roblox_cookie)

    if not csrf_token:
        print("❌ CSRF token retrieval failed.")
        return ""

    session.cookies.set(".ROBLOSECURITY", roblox_cookie)
    session.headers.update({
        "User-Agent": "RobloxLauncher",
        "Referer": "https://www.roblox.com/",
        "X-CSRF-TOKEN": csrf_token
    })

    response = session.post("https://auth.roblox.com/v1/authentication-ticket", json={})

    if response.status_code == 200 and "RBX-Authentication-Ticket" in response.headers:
        print("✅ Successfully retrieved authentication ticket.")
        authTicket = response.headers["RBX-Authentication-Ticket"]
        print(authTicket)
        global AuthTickets
        AuthTickets[roblox_cookie] = authTicket
        return authTicket

    print(f"❌ Response Code: {response.status_code}")
    print(f"❌ Response Content: {response.text}")
    print("Most likely the Cookie was incorrect.")
    return ""

def logout_roblox(auth_ticket, cookie):
    session = requests.Session()
    csrf_token = get_csrf_token(session, cookie)

    if not csrf_token:
        print("❌ CSRF Token for logout not found.")
        return

    session.cookies.set(".ROBLOSECURITY", cookie)
    session.headers.update({
        "User-Agent": "RobloxLauncher",
        "X-CSRF-TOKEN": csrf_token
    })

    response = session.post("https://auth.roblox.com/v1/logout", json={})

    if response.status_code == 200:
        print("✅ Successfully logged out.")
    else:
        print(f"❌ Logout failed. Response: {response.status_code}")
        return "Error"

def getData(placeId):
    try:
        response = requests.get(url=f"https://games.roblox.com/v1/games/{placeId}/servers/Public?sortOrder=Asc&excludeFullGames=true&limit=100")
        if response:
            return response.json()
        else:
            print("failed getting data from api")
    except Exception as e:
        print(f"error happened in getData function: {e}")

def sortData(response):
    global  SERVERLISTS
    serverList = []
    if response:
        try:
            for server in response["data"]:
                if server["maxPlayers"] -3 >= server["playing"]:
                    serverList.append(server)
                    SERVERLISTS = serverList

        except Exception as e:
            print(f"Erorr happened in sortData function: {e}")

def LoopSvr(placeId):
    response = getData(placeId)
    sortData(response=response)
    time.sleep(5 * 60)
def getSvr():
    svr = SERVERLISTS[0]
    SERVERLISTS.remove(SERVERLISTS[0])
    svrId = svr["id"]

def getTime():
    delay = None
    while True:
        try:
            user_input = input(
                "How long should we wait? (e.g., 30s, 5m, 1h): ").strip().lower()
            if user_input.endswith("s"):
                delay = int(user_input[:-1])
            elif user_input.endswith("m"):
                delay = int(user_input[:-1]) * 60
            elif user_input.endswith("h"):
                delay = int(user_input[:-1]) * 3600

            else:
                print("Invalid format. Use s for seconds, m for minutes, or h for hours.")
                continue

            break
        except Exception as e:
            print(f"Invalid input: {e}")
    return delay

def logout_all_instances():
    """
    Stops all running instances by logging out of all associated authentication tickets.
    """
    print("Stopping all instances...")
    for cookie, auth_ticket in AuthTickets.items():
        if cookie:
            response = logout_roblox(auth_ticket, cookie)
    if response == "Error":
        print("Error in logging out.")
        return
    AuthTickets.clear()
    print("All instances stopped.")

def logout_instance(cookie):
    """
    Stops the instance associated with a specific cookie.
    """
    auth_ticket = AuthTickets.get(cookie)
    if auth_ticket:
        print(f"Stopping instance for cookie: {cookie}")
        logout_roblox(auth_ticket, cookie)
        del AuthTickets[cookie]
        print(f"Instance for cookie {cookie} stopped.")
    else:
        print(f"No authentication ticket found for cookie {cookie}.")

def start_instance(cookie, placeId):
    """
    Starts a new instance with the provided cookie and adds it to the AuthTickets.
    """
    if cookie:
        print(f"Starting instance for cookie: {cookie}")
        LoginProcess(cookie, placeId)
    else:
        print("Cookie not provided.")

def change_script(new_script):
    """
    Changes the script being executed by the instances.
    Replaces the code inside the pcall block with new_script.
    """
    if new_script:
        print("Changing script...")
        path = "Xeno-v1.1.85/autoexec/script.txt"

        with open(path, "r") as f:
            script_content = f.read()

        # Find the start and end of the pcall block
        start_marker = "local success, message = pcall(function()\n"
        end_marker = "\nend)"

        start_index = script_content.find(start_marker)
        end_index = script_content.find(end_marker, start_index)

        if start_index == -1 or end_index == -1:
            raise Exception("Script markers not found.")

        # Rebuild script with the new content
        updated_script = (
            script_content[:start_index + len(start_marker)] +
            f"\t{new_script}\n" +
            script_content[end_index:]
        )

        with open(path, "w") as f:
            f.write(updated_script)
        print("Script successfully updated.")
    else:
        print("No script provided.")

def count_instances():
    return len(RUNNING_INSTANCES)


def stop_all():
    global RUNNING_INSTANCES
    if len(RUNNING_INSTANCES) == 0:
        print("There are no current running instances.")
        return

    # Attempt to terminate processes gracefully first
    for cookie, process in RUNNING_INSTANCES.items():
        if process:
            process.terminate()
            print(f"✅ Termination requested for Roblox instance associated with cookie {cookie}.")
            process.wait(timeout=1)  # Wait for a short duration to allow graceful shutdown
            if process.poll() is None:  # Process is still running
                print(f"❌ Process for cookie {cookie} did not terminate in time.")

    # After attempting graceful shutdown, force kill processes
    kill_all_roblox_processes()

    AuthTickets.clear()  # Clear the authentication tickets
    RUNNING_INSTANCES.clear()  # Clear the running instances
    print("All instances stopped.")
def kill_all_roblox_processes():

        print("✅ All Roblox processes terminated.")
    except Exception as e:
        print(f"❌ Failed to kill processes: {e}")
